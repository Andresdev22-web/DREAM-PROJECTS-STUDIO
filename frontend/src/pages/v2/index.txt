    <script>
      const menuButton = document.querySelector<HTMLButtonElement>("#mobile-menu");
      const mobileNav = document.querySelector<HTMLElement>("#mobile-nav");

      menuButton?.addEventListener("click", () => {
        if (!mobileNav) return;

        const isExpanded = menuButton.getAttribute("aria-expanded") === "true";

        menuButton.setAttribute("aria-expanded", String(!isExpanded));
        menuButton.setAttribute(
          "aria-label",
          isExpanded ? "Abrir menú" : "Cerrar menú",
        );
        mobileNav.hidden = isExpanded;

        const icon = menuButton.querySelector("svg");
        icon?.setAttribute(
          "data-lucide-state",
          isExpanded ? "menu" : "close",
        );
      });

      mobileNav?.querySelectorAll("a").forEach((link) => {
        link.addEventListener("click", () => {
          mobileNav.hidden = true;
          menuButton?.setAttribute("aria-expanded", "false");
          menuButton?.setAttribute("aria-label", "Abrir menú");
        });
      });

      const demoForm = document.querySelector<HTMLFormElement>("#demo-form");
      const feedback = document.querySelector<HTMLElement>("#form-feedback");

      demoForm?.addEventListener("submit", (event) => {
        event.preventDefault();

        if (!demoForm.reportValidity()) return;

        const formData = new FormData(demoForm);
        const name = String(formData.get("name") || "").trim();

        if (feedback) {
          feedback.textContent = `¡Perfecto, ${name}! La validación funciona correctamente. Este formulario no envía datos a ningún servidor.`;
          feedback.classList.add("feedback-success");
        }
      });
    </script>
